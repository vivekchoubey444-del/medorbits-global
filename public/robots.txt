# payment endpoints, APIs, and confidential files using server-side
# authentication, authorization, and appropriate storage permissions.
#
# DEPLOYMENT REQUIREMENTS
# ---------------------------------------------------------------------
# 1. Confirm that https://medorbits.in is the production canonical host.
# 2. Ensure this file is served as plain text at /robots.txt.
# 3. Confirm every rule against the actual deployed URL structure.
# 4. Keep important public resources crawlable.
# 5. Do not block resources required to render public pages.
# 6. Avoid blocking URLs solely to prevent their appearance in search
#    results; robots.txt is not a reliable deindexing mechanism.
# 7. Test rules using the relevant search engine's testing tools.

# =====================================================================
# 09. INTERNAL SEARCH AND QUERY-PARAMETER CRAWL CONTROL
# =====================================================================
#
# These patterns are primarily intended for crawlers that support
# wildcard matching in robots.txt. Query-string handling varies among
# crawlers. Test before deployment.
#
# Do not block parameters that generate unique, useful public content.
# Use canonical tags to consolidate duplicate public URLs where suitable.
# Keep private tokens out of URLs wherever possible.
#
# Internal search routes.
# The wildcard form matches /search?query=... and /search/?query=...
# on crawlers supporting the wildcard syntax.

Disallow: /search?*
Disallow: /search/*?*

# Common sorting and ordering query variants.
# These patterns are not a universal query-parameter policy.

Disallow: /*?sort=*
Disallow: /*&sort=*
Disallow: /*?order=*
Disallow: /*&order=*

# Session and authentication query parameters.
# Avoid placing secrets in URLs; robots.txt cannot protect their values.

Disallow: /*?session=*
Disallow: /*&session=*
Disallow: /*?sessionid=*
Disallow: /*&sessionid=*
Disallow: /*?token=*
Disallow: /*&token=*
Disallow: /*?access_token=*
Disallow: /*&access_token=*
Disallow: /*?auth=*
Disallow: /*&auth=*
Disallow: /*?password=*
Disallow: /*&password=*
Disallow: /*?verification=*
Disallow: /*&verification=*
Disallow: /*?reset_token=*
Disallow: /*&reset_token=*
